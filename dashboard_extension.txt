<style>
  /* Namespace: bdFlip — 1-bit B/W Flip Clock */
  .bdflip {
    width: 300px;
  }
  .bdflip .host {
    width: 100%;
    background: var(--bd-bg, #fff);
    box-sizing: border-box;
    display: flex;
    align-items: center;
    justify-content: center;
    overflow: hidden;
  }
  .bdflip svg {
    display: block;
    width: 100%;
    height: auto;
    overflow: hidden;
  }

<script>
  /* Namespace: bdFlip — themed flip clock, 4-bit auto-shading */
  (function () {
    const NS = 'http://www.w3.org/2000/svg';
    // Terminus has no TRMNL user settings, so these are fixed to the plugin defaults.
    const GLOBAL_IS_24H = true;
    const THEME_NAME = 'Classic';
    const SHOW_BORDER = true;

    const is4bit = !!document.querySelector('.screen--4bit');
    const isGray3D = THEME_NAME === 'Grayscale 3D';

    const THEMES = {
      classic: {
        tileFill: '#000',
        tileStroke: '#FFF',
        strokeW: 6,
        digitFill: '#FFF',
        digitStroke: '#000',
        digitStrokeW: 6,
        seamColor: '#FFF',
        seamW: 2,
        bgColor: '#FFF',
        shadow: null,
        shadowOffset: 0,
        seamShadow: null,
        shadow4bit: '#AAA',
        shadowOffset4bit: 2,
        seamShadow4bit: '#CCC',
        bg4bit: '#F5F5F5',
        bevelLight: null,
        bevelDark: null,
        splitFill: false,
      },
      grayscale3d: {
        tileFill: '#000',
        tileFillTop: '#1A1A1A',
        tileFillBot: '#000',
        tileStroke: '#CCC',
        strokeW: 5,
        digitFill: '#FFF',
        digitStroke: '#222',
        digitStrokeW: 6,
        seamColor: '#DDD',
        seamW: 2,
        seamShadow: '#333',
        bgColor: '#E8E8E8',
        shadow: '#888',
        shadowOffset: 3,
        bevelLight: '#555',
        bevelDark: '#111',
        splitFill: true,
      },
    };

    const themeKey = isGray3D ? 'grayscale3d' : 'classic';
    const T = THEMES[themeKey];

    const resolvedBg = themeKey === 'classic' && is4bit && T.bg4bit ? T.bg4bit : T.bgColor;
    const resolvedShadow = T.shadow || (is4bit && T.shadow4bit) || null;
    const resolvedShadowOffset = T.shadowOffset || (is4bit && T.shadowOffset4bit) || 0;
    const resolvedSeamShadow = T.seamShadow || (is4bit && T.seamShadow4bit) || null;

    const STROKE = T.strokeW;
    const DESIGN = { tileW: 120, tileH: 200, gap: 10, r: 22, font: 128 };

    function boot() {
      document.querySelectorAll('[data-bdflip="clock"]').forEach((el) => {
        if (el._bdBoot) return;
        el._bdBoot = true;
        build(el);
      });
    }

    function build(el) {
      el.innerHTML = '<div class="host"></div>';
      const host = el.querySelector('.host');
      host.style.setProperty('--bd-bg', resolvedBg);
      const svg = document.createElementNS(NS, 'svg');
      svg.setAttribute('preserveAspectRatio', 'xMidYMid meet');
      host.appendChild(svg);
      el._svg = svg;
      render(el);
      schedule(el);

      new MutationObserver(() => render(el)).observe(el, {
        attributes: true,
        attributeFilter: ['data-12or24', 'data-ampm-scale'],
      });
    }

    function is24(el) {
      const o = (el.getAttribute('data-12or24') || '').trim();
      return o ? o === '24' : GLOBAL_IS_24H;
    }

    // Uses the renderer's local clock; Terminus has no per-user IANA zone or UTC offset.
    function timeParts(use24) {
      const d = new Date();
      let h = d.getHours(),
        m = d.getMinutes(),
        suf = null;
      if (!use24) {
        suf = h >= 12 ? ['P', 'M'] : ['A', 'M'];
        h = h % 12 || 12;
      }
      const HH = String(h).padStart(2, '0');
      const MM = String(m).padStart(2, '0');
      return { digits: [HH[0], HH[1], MM[0], MM[1]], suffix: suf };
    }

    function rect(x, y, w, h, r, fill, stroke, sw) {
      const e = document.createElementNS(NS, 'rect');
      e.setAttribute('x', x);
      e.setAttribute('y', y);
      e.setAttribute('width', w);
      e.setAttribute('height', h);
      if (r != null) e.setAttribute('rx', r);
      if (fill != null) e.setAttribute('fill', fill);
      if (stroke) {
        e.setAttribute('stroke', stroke);
        e.setAttribute('stroke-width', sw || 1);
        e.setAttribute('stroke-linejoin', 'round');
      }
      return e;
    }

    function line(x1, y1, x2, y2, stroke, sw) {
      const l = document.createElementNS(NS, 'line');
      l.setAttribute('x1', x1);
      l.setAttribute('y1', y1);
      l.setAttribute('x2', x2);
      l.setAttribute('y2', y2);
      l.setAttribute('stroke', stroke);
      l.setAttribute('stroke-width', sw || 1);
      return l;
    }

    function makeTile(x) {
      const g = document.createElementNS(NS, 'g');
      g.setAttribute('transform', `translate(${x},0)`);
      const W = DESIGN.tileW,
        H = DESIGN.tileH,
        inset = STROKE / 2;

      if (resolvedShadow) {
        const so = resolvedShadowOffset;
        g.appendChild(rect(so, so, W, H, DESIGN.r, resolvedShadow));
      }

      if (T.splitFill) {
        const clipId = 'ct' + Math.random().toString(36).slice(2, 8);
        const clipIdB = 'cb' + Math.random().toString(36).slice(2, 8);
        const defs = document.createElementNS(NS, 'defs');

        const clipTop = document.createElementNS(NS, 'clipPath');
        clipTop.setAttribute('id', clipId);
        clipTop.appendChild(rect(0, 0, W, H / 2, 0, null));
        defs.appendChild(clipTop);

        const clipBot = document.createElementNS(NS, 'clipPath');
        clipBot.setAttribute('id', clipIdB);
        clipBot.appendChild(rect(0, H / 2, W, H / 2, 0, null));
        defs.appendChild(clipBot);

        g.appendChild(defs);

        const topRect = rect(0, 0, W, H, DESIGN.r, T.tileFillTop);
        topRect.setAttribute('clip-path', `url(#${clipId})`);
        g.appendChild(topRect);

        const botRect = rect(0, 0, W, H, DESIGN.r, T.tileFillBot);
        botRect.setAttribute('clip-path', `url(#${clipIdB})`);
        g.appendChild(botRect);
      } else {
        g.appendChild(rect(0, 0, W, H, DESIGN.r, T.tileFill));
      }

      if (SHOW_BORDER) {
        g.appendChild(
          rect(inset, inset, W - STROKE, H - STROKE, Math.max(0, DESIGN.r - inset), 'none', T.tileStroke, STROKE)
        );
      }

      const t = document.createElementNS(NS, 'text');
      t.setAttribute('x', W / 2);
      t.setAttribute('y', H / 2 + DESIGN.font * 0.35);
      t.setAttribute('text-anchor', 'middle');
      t.setAttribute('font-family', 'Inter,system-ui,Arial,sans-serif');
      t.setAttribute('font-weight', '900');
      t.setAttribute('font-size', DESIGN.font);
      t.setAttribute('fill', T.digitFill);
      t.setAttribute('style', `paint-order:stroke;stroke:${T.digitStroke};stroke-width:${T.digitStrokeW};`);
      g.appendChild(t);

      const seamX1 = inset * 2,
        seamX2 = W - inset * 2;
      g.appendChild(line(seamX1, H / 2, seamX2, H / 2, T.seamColor, T.seamW));

      if (resolvedSeamShadow) {
        g.appendChild(line(seamX1, H / 2 + T.seamW, seamX2, H / 2 + T.seamW, resolvedSeamShadow, 1));
      }

      const overlay = document.createElementNS(NS, 'g');
      overlay.setAttribute('class', 'ampm');
      g.appendChild(overlay);

      return { g, text: t, overlay, w: W };
    }

    // Center horizontally, sit near the bottom of the last tile
    function drawAmPm(overlay, label, scale) {
      while (overlay.firstChild) overlay.removeChild(overlay.firstChild);

      const W = DESIGN.tileW,
        H = DESIGN.tileH;

      // size (same as before, default 0.33), and bottom inset in px (default 12)
      const s = Math.max(0.2, Math.min(1, parseFloat(scale) || 0.33));
      const inset = parseFloat(overlay.closest('[data-bdflip="clock"]').getAttribute('data-ampm-inset')) || 12;

      // place at bottom-center, with a little safety so it never touches the outline
      const g = document.createElementNS(NS, 'g');
      g.setAttribute('transform', `translate(${W / 2}, ${H - inset}) scale(${s})`);

      const t = document.createElementNS(NS, 'text');
      t.setAttribute('x', 0);
      t.setAttribute('y', 0);
      t.setAttribute('text-anchor', 'middle'); // horizontal center
      t.setAttribute('dominant-baseline', 'alphabetic'); // baseline sits on the bottom inset line
      t.setAttribute('font-family', 'Inter,system-ui,Arial,sans-serif');
      t.setAttribute('font-weight', '900');
      t.setAttribute('font-size', 90);
      t.setAttribute('letter-spacing', '0.04em');
      t.setAttribute('fill', T.digitFill);
      t.setAttribute('style', `paint-order:stroke;stroke:${T.digitStroke};stroke-width:12;`);
      t.textContent = label;

      g.appendChild(t);
      overlay.appendChild(g);
    }

    function render(el) {
      const use24 = is24(el);
      const { digits, suffix } = timeParts(use24);
      const ampmScale = parseFloat(el.getAttribute('data-ampm-scale') || '0.33');
      const n = digits.length;
      const so = resolvedShadowOffset;
      const totalW = n * DESIGN.tileW + (n - 1) * DESIGN.gap + so;
      const totalH = DESIGN.tileH + so;
      const svg = el._svg;
      svg.setAttribute('viewBox', `0 0 ${totalW} ${totalH}`);

      while (svg.firstChild) svg.removeChild(svg.firstChild);
      let x = 0;
      el._texts = [];
      const tileGroups = [];

      for (let i = 0; i < n; i++) {
        const { g, text, overlay, w } = makeTile(x);
        svg.appendChild(g);
        text.textContent = digits[i];
        el._texts.push(text);
        tileGroups.push({ g, overlay });
        x += w + (i < n - 1 ? DESIGN.gap : 0);
      }

      if (suffix) {
        const last = tileGroups[tileGroups.length - 1];
        drawAmPm(last.overlay, suffix.join(''), ampmScale);
      }
    }

    function schedule(el) {
      (function atMinute() {
        const now = new Date();
        const ms = (60 - now.getSeconds()) * 1000 - now.getMilliseconds();
        setTimeout(() => {
          const use24 = is24(el);
          const { digits, suffix } = timeParts(use24);
          for (let i = 0; i < el._texts.length; i++) {
            if (digits[i]) el._texts[i].textContent = digits[i];
          }
          const ampmScale = parseFloat(el.getAttribute('data-ampm-scale') || '0.33');
          const overlays = el._svg.querySelectorAll('g.ampm');
          overlays.forEach((o) => {
            while (o.firstChild) o.removeChild(o.firstChild);
          });
          if (suffix && overlays.length) {
            drawAmPm(overlays[overlays.length - 1], suffix.join(''), ampmScale);
          }
          atMinute();
        }, Math.max(200, ms));
      })();
    }

    if (document.readyState === 'loading') document.addEventListener('DOMContentLoaded', boot);
    else boot();
  })();
</script>
